Korea_and_Japan = [
    "data_quiz_1/korea_rep_co2_emissions.csv",
   "data_quiz_1/japan_co2_emissions.csv"]

list_emissions = []

# Compute mean of 'total_emissions' for each file
for file in Korea_and_Japan:
    df = pd.read_csv(file)
    mean_val = df["total_emissions"].mean()
    list_emissions.append(mean_val)
